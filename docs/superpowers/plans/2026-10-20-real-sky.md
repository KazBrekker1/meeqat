# Real Sky Behind the Moon — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the home screen's random star-field with the real sky around the Moon (real positions, always dark, ground dimmed below the horizon), and let the user tap the Moon to explore the whole sky with search, a prayer-time slider and a Qibla marker.

**Architecture:** Pure TypeScript modules under `app/utils/sky/` do the astronomy, the catalogue format, the projection, picking, search and the info-card text; they are unit-tested with `bun test` against astropy reference positions. A WebGL 1 renderer draws the sky, constellation figures and stars in three draw calls from a catalogue uploaded once; a 2D overlay canvas draws text and markers. `RealSky.vue` draws only when something changes (minute tick, scroll, resize, input). Explore mode is the same canvas made interactive, driven by shared state (`useSkyExplore`) and route state (`?sky=1`).

**Tech Stack:** Nuxt 4 / Vue 3 (`<script setup>`, auto-imports), TypeScript, WebGL 1, Bun 1.4 (`bun test`, build script), astropy 8 via `uv` (reference data only), Playwright (one optional check).

**Spec:** `docs/superpowers/specs/2026-10-20-real-sky-design.md` · **Branch:** `feat/real-sky` (already created; the spec is committed on it).

```ts
// Small vector helpers for the sky. Directions are unit 3-vectors; matrices are row-major 3×3.
// Horizon frame: x east, y north, z up. Equatorial frame: x toward RA 0h, z toward the north pole.

export type Vec3 = [number, number, number];
export type Mat3 = [number, number, number, number, number, number, number, number, number];

export const RAD = Math.PI / 180;
export const TAU = Math.PI * 2;

export const clamp = (x: number, a: number, b: number) => Math.min(b, Math.max(a, x));
/** Angle wrapped to [0, 2π). */
export const wrapTau = (a: number) => ((a % TAU) + TAU) % TAU;
/** Angle wrapped to (−π, π]. */
export const wrapPi = (a: number) => Math.atan2(Math.sin(a), Math.cos(a));

export function vecFromRaDec(ra: number, dec: number): Vec3 {
  const c = Math.cos(dec);
  return [c * Math.cos(ra), c * Math.sin(ra), Math.sin(dec)];
}

/** Horizon direction for an altitude and an azimuth (from north, eastward), radians. */
export function vecFromAltAz(alt: number, az: number): Vec3 {
  const c = Math.cos(alt);
  return [c * Math.sin(az), c * Math.cos(az), Math.sin(alt)];
}

export function altAzOf(d: ArrayLike<number>, o = 0): { alt: number; az: number } {
  return { alt: Math.asin(clamp(d[o + 2]!, -1, 1)), az: wrapTau(Math.atan2(d[o]!, d[o + 1]!)) };
}

export function mulMat(a: Mat3, b: Mat3): Mat3 {
  const m = new Array(9).fill(0) as Mat3;
  for (let i = 0; i < 3; i++)
    for (let j = 0; j < 3; j++) m[i * 3 + j] = a[i * 3]! * b[j]! + a[i * 3 + 1]! * b[3 + j]! + a[i * 3 + 2]! * b[6 + j]!;
  return m;
}

/** m · v, reading v from `o` (so it works on packed Float32Arrays too). */
export function mulVec(m: Mat3, v: ArrayLike<number>, o = 0): Vec3 {
  const x = v[o]!, y = v[o + 1]!, z = v[o + 2]!;
  return [m[0] * x + m[1] * y + m[2] * z, m[3] * x + m[4] * y + m[5] * z, m[6] * x + m[7] * y + m[8] * z];
}

/** Angle between two unit vectors, radians (stable for tiny angles). */
export function angleBetween(a: Vec3, b: Vec3): number {
  const cx = a[1] * b[2] - a[2] * b[1], cy = a[2] * b[0] - a[0] * b[2], cz = a[0] * b[1] - a[1] * b[0];
  return Math.atan2(Math.hypot(cx, cy, cz), a[0] * b[0] + a[1] * b[1] + a[2] * b[2]);
}

/** Column-major copy for gl.uniformMatrix3fv. */
export function toColumnMajor(m: Mat3, out = new Float32Array(9)): Float32Array {
  out[0] = m[0]; out[1] = m[3]; out[2] = m[6];
  out[3] = m[1]; out[4] = m[4]; out[5] = m[7];
  out[6] = m[2]; out[7] = m[5]; out[8] = m[8];
  return out;
}
```

```ts
// The five naked-eye planets from JPL's "Approximate Positions of the Planets" (E. M. Standish,
// Table 1: Keplerian elements valid 1800–2050, a few arcminutes at worst), geocentric, J2000.
// Magnitudes: Explanatory Supplement (1992) phase laws; Saturn is the globe only (the rings add
// up to about a magnitude when they are open).
import { RAD, clamp, type Vec3 } from "./vec";

export type PlanetKey = "mercury" | "venus" | "mars" | "jupiter" | "saturn";

export interface Planet {
  key: PlanetKey;
  en: string;
  ar: string;
  /** apparent visual magnitude */
  mag: number;
  /** geocentric J2000 equatorial unit vector */
  dirJ2000: Vec3;
}

// a [au], ȧ, e, ė, I [°], İ, L [°], L̇, ϖ [°], ϖ̇, Ω [°], Ω̇ — rates per Julian century; J2000 ecliptic.
type Elements = readonly [number, number, number, number, number, number, number, number, number, number, number, number];

const EM_BARYCENTRE: Elements = [1.00000261, 0.00000562, 0.01671123, -0.00004392, -0.00001531, -0.01294668, 100.46457166, 35999.37244981, 102.93768193, 0.32327364, 0, 0];

export const PLANETS: readonly { key: PlanetKey; en: string; ar: string; el: Elements; phaseMag: (i: number) => number }[] = [
  { key: "mercury", en: "Mercury", ar: "عطارد", el: [0.38709927, 0.00000037, 0.20563593, 0.00001906, 7.00497902, -0.00594749, 252.2503235, 149472.67411175, 77.45779628, 0.16047689, 48.33076593, -0.12534081], phaseMag: (i) => -0.42 + 0.038 * i - 0.000273 * i * i + 0.000002 * i ** 3 },
  { key: "venus", en: "Venus", ar: "الزهرة", el: [0.72333566, 0.0000039, 0.00677672, -0.00004107, 3.39467605, -0.0007889, 181.9790995, 58517.81538729, 131.60246718, 0.00268329, 76.67984255, -0.27769418], phaseMag: (i) => -4.4 + 0.0009 * i + 0.000239 * i * i - 0.00000065 * i ** 3 },
  { key: "mars", en: "Mars", ar: "المريخ", el: [1.52371034, 0.00001847, 0.0933941, 0.00007882, 1.84969142, -0.00813131, -4.55343205, 19140.30268499, -23.94362959, 0.44441088, 49.55953891, -0.29257343], phaseMag: (i) => -1.52 + 0.016 * i },
  { key: "jupiter", en: "Jupiter", ar: "المشتري", el: [5.202887, -0.00011607, 0.04838624, -0.00013253, 1.30439695, -0.00183714, 34.39644051, 3034.74612775, 14.72847983, 0.21252668, 100.47390909, 0.20469106], phaseMag: (i) => -9.4 + 0.005 * i },
  { key: "saturn", en: "Saturn", ar: "زحل", el: [9.53667594, -0.0012506, 0.05386179, -0.00050991, 2.48599187, 0.00193609, 49.95424423, 1222.49362201, 92.59887831, -0.41897216, 113.66242448, -0.28867794], phaseMag: () => -8.88 },
];

const OBLIQUITY_J2000 = 23.43928 * RAD;

/** Heliocentric J2000 ecliptic position, au. */
function heliocentric(el: Elements, T: number): Vec3 {
  const a = el[0] + el[1] * T, e = el[2] + el[3] * T, I = (el[4] + el[5] * T) * RAD;
  const L = el[6] + el[7] * T, w = el[8] + el[9] * T, O = el[10] + el[11] * T;
  const om = (w - O) * RAD, Om = O * RAD;
  let M = (((L - w) % 360) + 360) % 360;
  if (M > 180) M -= 360;
  M *= RAD;
  let E = M + e * Math.sin(M);
  for (let k = 0; k < 6; k++) E -= (E - e * Math.sin(E) - M) / (1 - e * Math.cos(E));
  const xp = a * (Math.cos(E) - e), yp = a * Math.sqrt(1 - e * e) * Math.sin(E);
  const co = Math.cos(om), so = Math.sin(om), cO = Math.cos(Om), sO = Math.sin(Om), cI = Math.cos(I), sI = Math.sin(I);
  return [
    (co * cO - so * sO * cI) * xp + (-so * cO - co * sO * cI) * yp,
    (co * sO + so * cO * cI) * xp + (-so * sO + co * cO * cI) * yp,
    so * sI * xp + co * sI * yp,
  ];
}

/** Geocentric directions and magnitudes at a Julian Day (UT is fine at this accuracy). */
export function planetsAt(jd: number): Planet[] {
  const T = (jd - 2451545) / 36525;
  const earth = heliocentric(EM_BARYCENTRE, T), R = Math.hypot(...earth);
  const ce = Math.cos(OBLIQUITY_J2000), se = Math.sin(OBLIQUITY_J2000);
  return PLANETS.map(({ key, en, ar, el, phaseMag }) => {
    const p = heliocentric(el, T);
    const x = p[0] - earth[0], y = p[1] - earth[1], z = p[2] - earth[2];
    const r = Math.hypot(...p), delta = Math.hypot(x, y, z);
    const i = Math.acos(clamp((r * r + delta * delta - R * R) / (2 * r * delta), -1, 1)) / RAD;
    const ye = y * ce - z * se, ze = y * se + z * ce;
    return { key, en, ar, mag: phaseMag(i) + 5 * Math.log10(r * delta), dirJ2000: [x / delta, ye / delta, ze / delta] };
  });
}
```

```ts
// The sky for a place and instant: one rotation that takes catalogue (J2000) directions into the
// local horizon frame, plus the Moon, the Sun and the planets as horizon directions.
// Not modelled (all < 0.02°): nutation, aberration, proper motion, refraction.
import { julianDay, sunEcliptic, eclToEq, horizontal, moonView, type MoonView } from "../moon/sphere";
import { planetsAt, type Planet } from "./planets";
import { RAD, mulMat, mulVec, vecFromAltAz, type Mat3, type Vec3 } from "./vec";

const ARCSEC = RAD / 3600;

/** J2000 mean equator → mean equator of date (IAU 1976 precession; Meeus 21.2–21.4). */
export function precessionMatrix(jd: number): Mat3 {
  const T = (jd - 2451545) / 36525;
  const zeta = (2306.2181 * T + 0.30188 * T * T + 0.017998 * T ** 3) * ARCSEC;
  const z = (2306.2181 * T + 1.09468 * T * T + 0.018203 * T ** 3) * ARCSEC;
  const theta = (2004.3109 * T - 0.42665 * T * T - 0.041833 * T ** 3) * ARCSEC;
  const cz = Math.cos(zeta), sz = Math.sin(zeta), cZ = Math.cos(z), sZ = Math.sin(z), ct = Math.cos(theta), st = Math.sin(theta);
  return [
    cz * cZ * ct - sz * sZ, -sz * cZ * ct - cz * sZ, -cZ * st,
    cz * sZ * ct + sz * cZ, -sz * sZ * ct + cz * cZ, -sZ * st,
    cz * st, -sz * st, ct,
  ];
}

/** Greenwich mean sidereal time, radians (the expression sphere.ts `horizontal` uses). */
export function gmst(jd: number): number {
  const d = (280.46061837 + 360.98564736629 * (jd - 2451545)) % 360;
  return (d < 0 ? d + 360 : d) * RAD;
}

/** Equator of date → local horizon frame (x east, y north, z up) at a place (degrees, east +). */
export function equatorToHorizon(jd: number, lat: number, lng: number): Mat3 {
  const th = gmst(jd) + lng * RAD, phi = lat * RAD;
  const ct = Math.cos(th), st = Math.sin(th), cp = Math.cos(phi), sp = Math.sin(phi);
  return [
    -st, ct, 0,
    -sp * ct, -sp * st, cp,
    cp * ct, cp * st, sp,
  ];
}

export interface SkyState {
  /** J2000 equatorial → local horizon (x east, y north, z up). */
  rot: Mat3;
  /** The Moon as seen from the place (zenith-up tilt), with its horizon direction. */
  moon: MoonView & { dir: Vec3 };
  sun: { dir: Vec3; altDeg: number };
  planets: (Planet & { dir: Vec3 })[];
}

export function skyAt(date: Date, lat: number, lng: number): SkyState {
  const jd = julianDay(date), T = (jd - 2451545) / 36525;
  const rot = mulMat(equatorToHorizon(jd, lat, lng), precessionMatrix(jd));
  const mv = moonView(date, lat, lng);
  const s = sunEcliptic(T), se = eclToEq(s.lambda, 0, s.eps), sh = horizontal(jd, se.ra, se.dec, lat, lng);
  return {
    rot,
    moon: { ...mv, dir: vecFromAltAz(mv.altitude, mv.azimuth) },
    sun: { dir: vecFromAltAz(sh.altitude, sh.azimuth), altDeg: sh.altitude / RAD },
    planets: planetsAt(jd).map((p) => ({ ...p, dir: mulVec(rot, p.dirJ2000) })),
  };
}
```

```ts
// The star catalogue: a compact binary (stars + constellation figures) and a names file, built by
// scripts/sky/build-catalog.ts from d3-celestial and served from public/sky (immutable; a format
// change gets a new file name). Decoded once per page and shared.
//
// sky-v1.bin, little-endian:
//   0   "MQSK"          magic
//   4   u16 version     1
//   6   u16 stars
//   8   u16 vertices
//   10  u16 segments
//   12  stars × 9 B     i16 x, i16 y, i16 z (J2000 unit vector × 32767), i8 magnitude × 10,
//                       i8 B−V × 50, u8 constellation index (255 = unknown)
//   ..  vertices × 6 B  i16 x, i16 y, i16 z (figure vertex, J2000 unit vector × 32767)
//   ..  segments × 5 B  u16 a, u16 b (vertex indices), u8 constellation index
import { RAD, vecFromRaDec, type Vec3 } from "./vec";

export const SKY_BIN_URL = "/sky/sky-v1.bin";
export const SKY_NAMES_URL = "/sky/names-v1.json";
export const NO_CONSTELLATION = 255;

const MAGIC = "MQSK";
const VERSION = 1;
const HEADER = 12, STAR = 9, VERTEX = 6, SEGMENT = 5;
const Q = 32767;

/** names-v1.json */
export interface NamesFile {
  /** [star index, English, Arabic] for the named stars */
  stars: [number, string, string][];
  /** in figure order: the index here is the u8 constellation index in the binary */
  constellations: { id: string; en: string; ar: string; ra: number; dec: number; rank: number }[];
}

export interface Constellation {
  id: string;
  en: string;
  ar: string;
  /** where its name is written, J2000 unit vector */
  label: Vec3;
  /** 1 = major … 3 = minor */
  rank: number;
}

export interface SkyCatalog {
  /** J2000 equatorial unit vectors, 3 per star */
  starDirs: Float32Array;
  starMag: Float32Array;
  starBv: Float32Array;
  /** constellation index per star, NO_CONSTELLATION when unknown */
  starCons: Uint8Array;
  /** figure vertices, J2000 unit vectors, 3 per vertex */
  lineDirs: Float32Array;
  /** vertex index pairs, 2 per segment */
  segments: Uint16Array;
  /** constellation index per segment */
  segFigure: Uint8Array;
  names: { stars: { index: number; en: string; ar: string }[]; constellations: Constellation[] };
}

/** What the build script feeds the encoder (angles in degrees). */
export interface CatalogSource {
  stars: { ra: number; dec: number; mag: number; bv: number; cons: number }[];
  vertices: { ra: number; dec: number }[];
  segments: [a: number, b: number, cons: number][];
}

const q = (v: number) => Math.max(-Q, Math.min(Q, Math.round(v * Q)));

export function encodeSkyCatalog(src: CatalogSource): Uint8Array {
  const bytes = new Uint8Array(HEADER + src.stars.length * STAR + src.vertices.length * VERTEX + src.segments.length * SEGMENT);
  const dv = new DataView(bytes.buffer);
  for (let i = 0; i < 4; i++) bytes[i] = MAGIC.charCodeAt(i);
  dv.setUint16(4, VERSION, true);
  dv.setUint16(6, src.stars.length, true);
  dv.setUint16(8, src.vertices.length, true);
  dv.setUint16(10, src.segments.length, true);
  let o = HEADER;
  const putDir = (ra: number, dec: number) => {
    const d = vecFromRaDec(ra * RAD, dec * RAD);
    dv.setInt16(o, q(d[0]), true); dv.setInt16(o + 2, q(d[1]), true); dv.setInt16(o + 4, q(d[2]), true);
    o += 6;
  };
  for (const s of src.stars) {
    putDir(s.ra, s.dec);
    dv.setInt8(o, Math.round(s.mag * 10));
    dv.setInt8(o + 1, Math.round(Math.max(-0.5, Math.min(2.5, s.bv)) * 50));
    dv.setUint8(o + 2, s.cons);
    o += 3;
  }
  for (const v of src.vertices) putDir(v.ra, v.dec);
  for (const [a, b, c] of src.segments) {
    dv.setUint16(o, a, true); dv.setUint16(o + 2, b, true); dv.setUint8(o + 4, c);
    o += SEGMENT;
  }
  return bytes;
}

export function decodeSkyCatalog(buf: ArrayBuffer, names: NamesFile): SkyCatalog {
  const dv = new DataView(buf);
  const magic = String.fromCharCode(dv.getUint8(0), dv.getUint8(1), dv.getUint8(2), dv.getUint8(3));
  if (magic !== MAGIC || dv.getUint16(4, true) !== VERSION) throw new Error("sky catalogue: unknown format");
  const nStars = dv.getUint16(6, true), nVerts = dv.getUint16(8, true), nSegs = dv.getUint16(10, true);
  if (buf.byteLength !== HEADER + nStars * STAR + nVerts * VERTEX + nSegs * SEGMENT) throw new Error("sky catalogue: truncated");
  let o = HEADER;
  const getDir = (out: Float32Array, i: number) => {
    const x = dv.getInt16(o, true), y = dv.getInt16(o + 2, true), z = dv.getInt16(o + 4, true), n = Math.hypot(x, y, z);
    out[i * 3] = x / n; out[i * 3 + 1] = y / n; out[i * 3 + 2] = z / n;
    o += 6;
  };
  const starDirs = new Float32Array(nStars * 3), starMag = new Float32Array(nStars), starBv = new Float32Array(nStars), starCons = new Uint8Array(nStars);
  for (let i = 0; i < nStars; i++) {
    getDir(starDirs, i);
    starMag[i] = dv.getInt8(o) / 10;
    starBv[i] = dv.getInt8(o + 1) / 50;
    starCons[i] = dv.getUint8(o + 2);
    o += 3;
  }
  const lineDirs = new Float32Array(nVerts * 3);
  for (let i = 0; i < nVerts; i++) getDir(lineDirs, i);
  const segments = new Uint16Array(nSegs * 2), segFigure = new Uint8Array(nSegs);
  for (let i = 0; i < nSegs; i++) {
    segments[i * 2] = dv.getUint16(o, true);
    segments[i * 2 + 1] = dv.getUint16(o + 2, true);
    segFigure[i] = dv.getUint8(o + 4);
    o += SEGMENT;
  }
  return {
    starDirs, starMag, starBv, starCons, lineDirs, segments, segFigure,
    names: {
      stars: names.stars.map(([index, en, ar]) => ({ index, en, ar })),
      constellations: names.constellations.map((c) => ({ id: c.id, en: c.en, ar: c.ar, rank: c.rank, label: vecFromRaDec(c.ra * RAD, c.dec * RAD) })),
    },
  };
}

/** Promise cache that forgets a failure, so a failed load is retried next time. */
let pending: Promise<SkyCatalog> | null = null;
export function loadSkyCatalog(): Promise<SkyCatalog> {
  return (pending ??= (async () => {
    const [bin, names] = await Promise.all([
      fetch(SKY_BIN_URL).then((r) => (r.ok ? r.arrayBuffer() : Promise.reject(new Error(`${SKY_BIN_URL}: ${r.status}`)))),
      fetch(SKY_NAMES_URL).then((r) => (r.ok ? (r.json() as Promise<NamesFile>) : Promise.reject(new Error(`${SKY_NAMES_URL}: ${r.status}`)))),
    ]);
    return decodeSkyCatalog(bin, names);
  })().catch((err) => {
    pending = null;
    throw err;
  }));
}
```

```ts
// Builds public/sky/sky-v1.bin and public/sky/names-v1.json from d3-celestial (BSD-3) at a pinned
// commit. Run from the repo root: bun scripts/sky/build-catalog.ts
// Sources behind d3-celestial: stars from XHIP (VizieR V/137D), constellations from the IAU,
// star-name translations from Stellarium's sky cultures.
import { encodeSkyCatalog, NO_CONSTELLATION, type CatalogSource, type NamesFile } from "../../app/utils/sky/catalog";

const COMMIT = "7e720a3de062059d4c5400a379146a601d9010e0";
const FILES = {
  stars: ["stars.6.json", "0297b8fa3adfbce1dc26566f61c4abcc1df4f29c6a28729ca06b56d1c6d25602"],
  starnames: ["starnames.json", "19c84bc885f8a97c3b8e1f6a380084c575a9758dedfe35256e911a823ec3a695"],
  constellations: ["constellations.json", "ab4ae692027cbc042c0d6791a84456a65eb7c55656107fd00c58ff6e55d4d8b2"],
  lines: ["constellations.lines.json", "294f66bef5d5cf50b1e17f16d2efa1d97a15131612c68dd935adef6e7373e13c"],
} as const;
const MAX_MAG = 5.5;
const NAMED_MAX_MAG = 3.2;

async function fetchPinned(name: string, sha256: string): Promise<any> {
  const url = `https://raw.githubusercontent.com/ofrohn/d3-celestial/${COMMIT}/data/${name}`;
  const res = await fetch(url);
  if (!res.ok) throw new Error(`${url}: ${res.status}`);
  const bytes = new Uint8Array(await res.arrayBuffer());
  const got = new Bun.CryptoHasher("sha256").update(bytes).digest("hex");
  if (got !== sha256) throw new Error(`${name}: sha256 ${got}, expected ${sha256}`);
  return JSON.parse(new TextDecoder().decode(bytes));
}

/** d3-celestial stores right ascension as a longitude in −180…180. */
const ra = (lon: number) => (lon < 0 ? lon + 360 : lon);
/** Drop the bidi marks and odd spaces some names carry. */
const clean = (s: string | undefined) => (s ?? "").replace(/[‎‏]/g, "").replace(/[ - ]/g, " ").trim();

const [stars, starnames, constellations, lines] = await Promise.all(
  Object.values(FILES).map(([name, sha]) => fetchPinned(name, sha)),
);

const consList: { id: string }[] = constellations.features.map((f: any) => ({ id: f.id }));
const consIndex = (abbr: string | undefined) => {
  if (!abbr) return NO_CONSTELLATION;
  const i = consList.findIndex((c) => c.id === abbr || c.id.startsWith(abbr)); // "Ser" → Ser1
  return i < 0 ? NO_CONSTELLATION : i;
};

const src: CatalogSource = { stars: [], vertices: [], segments: [] };
const names: NamesFile = { stars: [], constellations: [] };

for (const f of stars.features) {
  const mag: number = f.properties.mag;
  if (mag > MAX_MAG) continue;
  const [lon, dec] = f.geometry.coordinates;
  const bv = Number.parseFloat(f.properties.bv);
  const n = starnames[String(f.id)];
  const index = src.stars.length;
  src.stars.push({ ra: ra(lon), dec, mag, bv: Number.isFinite(bv) ? bv : 0.6, cons: consIndex(n?.c) });
  if (n?.name && mag <= NAMED_MAX_MAG) names.stars.push([index, clean(n.name), clean(n.ar)]);
}

const vertexKey = new Map<string, number>();
const vertex = (lon: number, dec: number) => {
  const key = `${ra(lon).toFixed(4)},${dec.toFixed(4)}`;
  let i = vertexKey.get(key);
  if (i === undefined) {
    i = src.vertices.length;
    src.vertices.push({ ra: ra(lon), dec });
    vertexKey.set(key, i);
  }
  return i;
};
for (const f of lines.features) {
  const c = consList.findIndex((x) => x.id === f.id);
  if (c < 0) throw new Error(`figure ${f.id} has no constellation`);
  for (const line of f.geometry.coordinates as [number, number][][]) {
    for (let k = 1; k < line.length; k++) {
      const a = vertex(...line[k - 1]!), b = vertex(...line[k]!);
      if (a !== b) src.segments.push([a, b, c]);
    }
  }
}

for (const f of constellations.features) {
  const p = f.properties, [lon, dec] = f.geometry.coordinates;
  names.constellations.push({ id: f.id, en: clean(p.en || p.name), ar: clean(p.ar), ra: ra(lon), dec, rank: Number(p.rank) || 3 });
}

const bin = encodeSkyCatalog(src);
await Bun.write("public/sky/sky-v1.bin", bin);
await Bun.write("public/sky/names-v1.json", JSON.stringify(names));
console.log(
  `${src.stars.length} stars, ${names.stars.length} named, ${names.constellations.length} constellations, ` +
    `${src.vertices.length} figure vertices, ${src.segments.length} segments → sky-v1.bin ${bin.length} B`,
);
```

```ts
// Where the sky view looks and how directions land on the screen: a zenith-up camera with a
// stereographic projection (shapes stay true, so constellations look right even at 180°).
// Pixels are device pixels; the caller decides the canvas size and the centre point.
import { RAD, altAzOf, clamp, mulVec, wrapPi, wrapTau, type Mat3, type Vec3 } from "./vec";

export const FOV_MIN = 20;
export const FOV_MAX = 180;
/** Directions further than this behind the viewer are not drawn (they would project far off). */
const BEHIND = -0.8;

/** Explore mode's view: centre altitude and azimuth (radians) and the field of view across the width (degrees). */
export interface ViewState {
  alt: number;
  az: number;
  fovDeg: number;
}

export interface Camera {
  /** right, up and forward unit vectors in the horizon frame */
  r: Vec3;
  u: Vec3;
  f: Vec3;
  /** pixels per stereographic unit */
  k: number;
  /** where the view centre lands, and the canvas size, px */
  cx: number;
  cy: number;
  w: number;
  h: number;
}

/** Scale for a field of view `fovDeg` across `w` pixels, and back. */
export const kForFov = (fovDeg: number, w: number) => w / 2 / (2 * Math.tan((fovDeg * RAD) / 4));
export const fovForK = (k: number, w: number) => (4 * Math.atan(w / (4 * k))) / RAD;

/** A zenith-up camera looking at (alt, az). */
export function camera(alt: number, az: number, k: number, cx: number, cy: number, w: number, h: number): Camera {
  const ca = Math.cos(alt), sa = Math.sin(alt), cz = Math.cos(az), sz = Math.sin(az);
  return { f: [ca * sz, ca * cz, sa], r: [cz, -sz, 0], u: [-sz * sa, -cz * sa, ca], k, cx, cy, w, h };
}

/** Horizon direction (read from `d` at `o`) → pixels in out[oo], out[oo + 1]; false when behind the viewer. */
export function project(cam: Camera, d: ArrayLike<number>, out: number[] | Float32Array, o = 0, oo = 0): boolean {
  const x = d[o]!, y = d[o + 1]!, z = d[o + 2]!;
  const X = x * cam.r[0] + y * cam.r[1] + z * cam.r[2];
  const Y = x * cam.u[0] + y * cam.u[1] + z * cam.u[2];
  const Z = x * cam.f[0] + y * cam.f[1] + z * cam.f[2];
  if (Z < BEHIND) return false;
  const s = (2 / (1 + Z)) * cam.k;
  out[oo] = cam.cx + X * s;
  out[oo + 1] = cam.cy - Y * s;
  return true;
}

/** Pixels → horizon direction (unit vector). */
export function unproject(cam: Camera, px: number, py: number): Vec3 {
  const x = (px - cam.cx) / cam.k, y = (cam.cy - py) / cam.k, p2 = x * x + y * y;
  const Z = (4 - p2) / (4 + p2), h = (1 + Z) / 2, X = x * h, Y = y * h;
  return [
    X * cam.r[0] + Y * cam.u[0] + Z * cam.f[0],
    X * cam.r[1] + Y * cam.u[1] + Z * cam.f[1],
    X * cam.r[2] + Y * cam.u[2] + Z * cam.f[2],
  ];
}

export const onScreen = (cam: Camera, x: number, y: number, margin = 0) =>
  x > -margin && y > -margin && x < cam.w + margin && y < cam.h + margin;

/** Drag by (dx, dy) pixels on a canvas `w` pixels wide: the sky follows the pointer. */
export function pan(v: ViewState, dx: number, dy: number, w: number): ViewState {
  const perPx = (v.fovDeg * RAD) / w;
  return {
    alt: clamp(v.alt + dy * perPx, -89.5 * RAD, 89.5 * RAD),
    az: wrapTau(v.az - (dx * perPx) / Math.max(Math.cos(v.alt), 0.3)),
    fovDeg: v.fovDeg,
  };
}

/** Zoom by `factor` (< 1 zooms in) keeping the sky under pixel (x, y) where it is. */
export function zoomAt(v: ViewState, factor: number, x: number, y: number, cx: number, cy: number, w: number, h: number): ViewState {
  const fovDeg = clamp(v.fovDeg * factor, FOV_MIN, FOV_MAX);
  const want = altAzOf(unproject(camera(v.alt, v.az, kForFov(v.fovDeg, w), cx, cy, w, h), x, y));
  let next: ViewState = { alt: v.alt, az: v.az, fovDeg };
  for (let n = 0; n < 4; n++) {
    const got = altAzOf(unproject(camera(next.alt, next.az, kForFov(fovDeg, w), cx, cy, w, h), x, y));
    next = { alt: clamp(next.alt + want.alt - got.alt, -89.5 * RAD, 89.5 * RAD), az: wrapTau(next.az + wrapPi(want.az - got.az)), fovDeg };
  }
  return next;
}

/** Projects every J2000 direction in `dirs` through `rot` and the camera into `out` (x, y per item; NaN when hidden). */
export function projectAll(cam: Camera, rot: Mat3, dirs: Float32Array, out: Float32Array): Float32Array {
  const p = [0, 0];
  for (let i = 0, n = dirs.length / 3; i < n; i++) {
    const d = mulVec(rot, dirs, i * 3);
    if (project(cam, d, p) && onScreen(cam, p[0]!, p[1]!, 8)) {
      out[i * 2] = p[0]!;
      out[i * 2 + 1] = p[1]!;
    } else {
      out[i * 2] = out[i * 2 + 1] = Number.NaN;
    }
  }
  return out;
}

export type Hit = { kind: "body"; index: number } | { kind: "star"; index: number } | { kind: "segment"; index: number };

export interface PickInput {
  x: number;
  y: number;
  /** on-screen points that win over stars (planets, Moon, Sun, Qibla) */
  bodies: readonly { x: number; y: number }[];
  /** projected stars (projectAll) and their magnitudes */
  starXY: Float32Array;
  starMag: Float32Array;
  /** projected figure vertices (projectAll) and the vertex index pairs */
  vertXY: Float32Array;
  segments: Uint16Array;
  /** px: bodies within 1.6 × radius, stars within radius, segments within segRadius */
  radius: number;
  segRadius: number;
}

/** What is under (x, y): a body, else the best star (brighter wins ties), else the nearest figure segment. */
export function pick(p: PickInput): Hit | null {
  let best: Hit | null = null, bs = Infinity;
  p.bodies.forEach((b, index) => {
    const d = Math.hypot(b.x - p.x, b.y - p.y);
    if (d < p.radius * 1.6 && d < bs) { bs = d; best = { kind: "body", index }; }
  });
  if (best) return best;
  for (let i = 0; i < p.starMag.length; i++) {
    const sx = p.starXY[i * 2]!;
    if (Number.isNaN(sx)) continue;
    const d = Math.hypot(sx - p.x, p.starXY[i * 2 + 1]! - p.y);
    if (d > p.radius) continue;
    const score = d + p.starMag[i]! * (p.radius / 7);
    if (score < bs) { bs = score; best = { kind: "star", index: i }; }
  }
  if (best) return best;
  bs = p.segRadius;
  for (let i = 0; i < p.segments.length / 2; i++) {
    const a = p.segments[i * 2]!, b = p.segments[i * 2 + 1]!;
    const x1 = p.vertXY[a * 2]!, y1 = p.vertXY[a * 2 + 1]!, x2 = p.vertXY[b * 2]!, y2 = p.vertXY[b * 2 + 1]!;
    if (Number.isNaN(x1) || Number.isNaN(x2)) continue;
    const dx = x2 - x1, dy = y2 - y1, L = dx * dx + dy * dy || 1;
    const t = clamp(((p.x - x1) * dx + (p.y - y1) * dy) / L, 0, 1);
    const d = Math.hypot(x1 + t * dx - p.x, y1 + t * dy - p.y);
    if (d < bs) { bs = d; best = { kind: "segment", index: i }; }
  }
  return best;
}
```

```ts
// Search over everything explore mode can name: the named stars, the constellations, the planets,
// the Moon and the Sun, in English or Arabic.
import type { SkyCatalog } from "./catalog";
import { PLANETS, type PlanetKey } from "./planets";

export type SearchTarget =
  | { kind: "star"; index: number }
  | { kind: "constellation"; index: number }
  | { kind: "planet"; key: PlanetKey }
  | { kind: "moon" }
  | { kind: "sun" };

export interface SearchItem {
  target: SearchTarget;
  en: string;
  ar: string;
  /** ranking: brighter (smaller) first */
  mag: number;
  /** normalised English and Arabic names */
  keys: string[];
}

/**
 * Lower case, no Latin accents, no Arabic tashkeel or tatweel, and one letter for each of the
 * forms people type interchangeably: أ إ آ ٱ → ا, ى → ي, ة → ه.
 */
export function normalize(s: string): string {
  return s
    .normalize("NFKD")
    .replace(/\p{M}/gu, "")
    .replace(/ـ/g, "")
    .replace(/[آأإٱ]/g, "ا")
    .replace(/ى/g, "ي")
    .replace(/ة/g, "ه")
    .toLowerCase()
    .replace(/[\s -‏]+/g, " ")
    .trim();
}

/** Typical magnitudes, only for ranking. */
const PLANET_MAG: Record<PlanetKey, number> = { mercury: 0, venus: -4, mars: 1, jupiter: -2, saturn: 0.7 };
const RANK_MAG = [0, 1, 2.5, 4];

export function buildSearchIndex(cat: SkyCatalog): SearchItem[] {
  const item = (target: SearchTarget, en: string, ar: string, mag: number): SearchItem => ({
    target, en, ar, mag, keys: [normalize(en), normalize(ar)].filter(Boolean),
  });
  return [
    item({ kind: "moon" }, "Moon", "القمر", -12),
    item({ kind: "sun" }, "Sun", "الشمس", -26),
    ...PLANETS.map((p) => item({ kind: "planet", key: p.key }, p.en, p.ar, PLANET_MAG[p.key])),
    ...cat.names.stars.map((s) => item({ kind: "star", index: s.index }, s.en, s.ar, cat.starMag[s.index]!)),
    ...cat.names.constellations.map((c, index) => item({ kind: "constellation", index }, c.en, c.ar, RANK_MAG[c.rank] ?? 4)),
  ];
}

/** Best matches: whole-name prefix, then word prefix, then anywhere; brighter first within each. */
export function search(index: readonly SearchItem[], query: string, limit = 8): SearchItem[] {
  const q = normalize(query);
  if (!q) return [];
  const scored: { item: SearchItem; score: number }[] = [];
  for (const item of index) {
    let score = Infinity;
    for (const k of item.keys) {
      if (k.startsWith(q)) score = Math.min(score, 0);
      else if (k.split(" ").some((w) => w.startsWith(q))) score = Math.min(score, 1);
      else if (k.includes(q)) score = Math.min(score, 2);
    }
    if (score < Infinity) scored.push({ item, score });
  }
  scored.sort((a, b) => a.score - b.score || a.item.mag - b.item.mag);
  return scored.slice(0, limit).map((s) => s.item);
}
```

```ts
// constellation to highlight, and where to look.
import { compassPoint, distanceToKaabaKm, qiblaBearing } from "../qibla";
import type { SkyState } from "./astro";
import { NO_CONSTELLATION, type SkyCatalog } from "./catalog";
import type { PlanetKey } from "./planets";
import type { SearchTarget } from "./search";
import { RAD, altAzOf, mulVec, vecFromAltAz, type Vec3 } from "./vec";

export type SkyPick =
  | { kind: "star"; index: number }
  | { kind: "figure"; figure: number }
  | { kind: "planet"; key: PlanetKey }
  | { kind: "moon" }
  | { kind: "sun" }
  | { kind: "qibla" };

export interface SkyInfo {
  title: string;
  ar: string;
  /** "Star in Lyre", "Planet", … */
  kind: string;
  details: string[];
}

/** "37° up · WNW" or "12° below the horizon · NE" */
export function whereInSky(d: ArrayLike<number>): string {
  const { alt, az } = altAzOf(d);
  const a = Math.round(alt / RAD), dir = compassPoint(az / RAD);
  return a >= 0 ? `${a}° up · ${dir}` : `${-a}° below the horizon · ${dir}`;
}

const magnitude = (m: number) => `magnitude ${m.toFixed(1)}`;

export function describePick(p: SkyPick, sky: SkyState, cat: SkyCatalog, lat: number, lng: number): SkyInfo {
  switch (p.kind) {
    case "star": {
      const named = cat.names.stars.find((s) => s.index === p.index);
      const c = cat.starCons[p.index]!;
      const cons = c === NO_CONSTELLATION ? undefined : cat.names.constellations[c];
      return {
        title: named?.en ?? "Star",
        ar: named?.ar ?? "",
        kind: cons ? `Star in ${cons.en}` : "Star",
        details: [magnitude(cat.starMag[p.index]!), whereInSky(mulVec(sky.rot, cat.starDirs, p.index * 3))],
      };
    }
    case "figure": {
      const c = cat.names.constellations[p.figure]!;
      return { title: c.en, ar: c.ar, kind: "Constellation", details: [whereInSky(mulVec(sky.rot, c.label))] };
    }
    case "planet": {
      const pl = sky.planets.find((x) => x.key === p.key)!;
      return { title: pl.en, ar: pl.ar, kind: "Planet", details: [magnitude(pl.mag), whereInSky(pl.dir)] };
    }
    case "moon":
      return { title: "Moon", ar: "القمر", kind: "Moon", details: [`${Math.round(sky.moon.fraction * 100)}% lit`, whereInSky(sky.moon.dir)] };
    case "sun":
      return { title: "Sun", ar: "الشمس", kind: "Star", details: [whereInSky(sky.sun.dir)] };
    case "qibla": {
      const b = qiblaBearing(lat, lng);
      return {
```

```ts
// GLSL ES 1.0 (WebGL 1) for the sky. All passes share the zenith-up stereographic camera:
//   u_m   J2000 → camera (x right, y up, z forward) = camera basis · sky rotation
//   u_c   where the view centre lands (px), u_k pixels per stereographic unit, u_res canvas size (px)
// Colours are premultiplied (blend ONE, ONE_MINUS_SRC_ALPHA).

const PROJECT = `
uniform mat3 u_m;
uniform vec2 u_res;
uniform vec2 u_c;
uniform float u_k;
vec2 toPx(vec3 c) {
  float s = 2.0 / (1.0 + c.z) * u_k;
  return vec2(u_c.x + c.x * s, u_c.y - c.y * s);
}
vec4 toClip(vec2 px) {
  return vec4(px.x / u_res.x * 2.0 - 1.0, 1.0 - px.y / u_res.y * 2.0, 0.0, 1.0);
}
const vec4 HIDDEN = vec4(2.0, 2.0, 2.0, 1.0);
`;

const FRAG_PRECISION = `
#ifdef GL_FRAGMENT_PRECISION_HIGH
precision highp float;
#else
precision mediump float;
#endif
`;

/** Full-screen triangle: the night gradient by altitude and the ground below the horizon. */
export const SKY_VS = `
attribute vec2 a_pos;
void main() { gl_Position = vec4(a_pos, 0.0, 1.0); }
`;
export const SKY_FS = `${FRAG_PRECISION}
uniform vec2 u_res;
uniform vec2 u_c;
uniform float u_k;
uniform vec3 u_cup; // the horizon "up" component of the camera's x, y, z axes
const vec3 HORIZON = vec3(28.0, 30.0, 74.0) / 255.0;
const vec3 ZENITH = vec3(6.0, 9.0, 26.0) / 255.0;
const vec3 GROUND = vec3(13.0, 10.0, 20.0) / 255.0;
void main() {
  vec2 xy = vec2(gl_FragCoord.x - u_c.x, u_c.y - (u_res.y - gl_FragCoord.y)) / u_k;
  float p2 = dot(xy, xy);
  float z = (4.0 - p2) / (4.0 + p2);
  float h = (1.0 + z) * 0.5;
  float up = dot(vec3(xy * h, z), u_cup);
  float alt = degrees(asin(clamp(up, -1.0, 1.0)));
  vec3 c = mix(HORIZON, ZENITH, smoothstep(0.0, 65.0, max(alt, 0.0)));
  c = mix(c, GROUND, 0.62 * (1.0 - smoothstep(-6.0, 0.0, alt)));
  gl_FragColor = vec4(c, 1.0);
}
`;

/** One point per star: size and brightness from magnitude, colour from B−V, dimmed below the horizon. */
export const STAR_VS = `
attribute vec3 a_dir;
attribute vec2 a_magbv;
${PROJECT}
uniform vec3 u_hz;      // J2000 → horizon "up" (third row of the sky rotation)
uniform float u_scale;  // px per size unit
uniform float u_below;  // brightness below the horizon
varying vec3 v_col;
varying float v_a;
varying float v_r;
varying float v_glow;
varying float v_size;
vec3 bvColor(float bv) {
  vec3 blue = vec3(0.67, 0.76, 1.0), white = vec3(0.98, 0.97, 1.0), yellow = vec3(1.0, 0.91, 0.78), orange = vec3(1.0, 0.8, 0.59);
  if (bv < 0.4) return mix(blue, white, clamp((bv + 0.1) / 0.5, 0.0, 1.0));
  if (bv < 0.9) return mix(white, yellow, (bv - 0.4) / 0.5);
  return mix(yellow, orange, clamp((bv - 0.9) / 0.6, 0.0, 1.0));
}
void main() {
  vec3 c = u_m * a_dir;
  float mag = a_magbv.x;
  v_glow = mag < 1.9 ? 1.0 : 0.0;
  v_col = v_glow > 0.5 ? bvColor(a_magbv.y) : vec3(0.886, 0.91, 1.0);
  v_a = clamp(1.12 - 0.12 * (mag + 1.0), 0.32, 1.0) * (dot(u_hz, a_dir) < 0.0 ? u_below : 1.0);
  v_r = max(0.7, 1.9 - 0.25 * mag) * u_scale;
  v_size = 2.0 * (v_glow > 0.5 ? v_r * 2.8 : v_r) + 2.0;
  if (c.z < -0.8) { gl_Position = HIDDEN; gl_PointSize = 0.0; return; }
  gl_Position = toClip(toPx(c));
  gl_PointSize = v_size;
}
`;
export const STAR_FS = `${FRAG_PRECISION}
varying vec3 v_col;
varying float v_a;
varying float v_r;
varying float v_glow;
varying float v_size;
void main() {
  float d = length(gl_PointCoord - 0.5) * v_size;
  float core = clamp(v_r + 0.5 - d, 0.0, 1.0);
  float halo = v_glow * 0.22 * clamp(1.0 - d / (v_r * 2.8), 0.0, 1.0);
  float a = max(core, halo) * v_a;
  if (a < 0.004) discard;
  gl_FragColor = vec4(v_col * a, a);
}
`;

/** Constellation figures as quads of constant on-screen width, edge anti-aliased; one figure in amber. */
export const LINE_VS = `
attribute vec3 a_a;       // segment start (J2000)
attribute vec3 a_b;       // segment end
attribute vec3 a_meta;    // which end (0 = a, 1 = b), side (±1), constellation index
${PROJECT}
uniform vec3 u_hz;
uniform float u_width;    // px
uniform float u_fig;      // highlighted constellation, −1 for none
uniform float u_below;
varying float v_hi;
varying float v_dim;
varying float v_px;
varying float v_hw;
void main() {
  vec3 ca = u_m * a_a, cb = u_m * a_b;
  v_hi = abs(a_meta.z - u_fig) < 0.5 ? 1.0 : 0.0;
  v_dim = dot(u_hz, mix(a_a, a_b, a_meta.x)) < 0.0 ? u_below : 1.0;
  v_hw = u_width * (v_hi > 0.5 ? 1.8 : 1.0) * 0.5;
  v_px = a_meta.y * (v_hw + 1.0);
  vec2 p = toPx(ca), q = toPx(cb), d = q - p;
  float len = length(d);
  if (ca.z < -0.8 || cb.z < -0.8 || len < 0.001 || len > u_res.x * 0.6) { gl_Position = HIDDEN; return; }
  vec2 n = vec2(-d.y, d.x) / len;
  gl_Position = toClip(mix(p, q, a_meta.x) + n * v_px);
}
`;
export const LINE_FS = `${FRAG_PRECISION}
uniform vec4 u_color;
uniform vec4 u_hiColor;
varying float v_hi;
varying float v_dim;
varying float v_px;
varying float v_hw;
void main() {
  float cover = clamp(v_hw + 0.5 - abs(v_px), 0.0, 1.0);
  gl_FragColor = (v_hi > 0.5 ? u_hiColor : u_color * v_dim) * cover;
```

```ts
// The WebGL 1 sky: uploads the catalogue once, then each frame is three draws (sky, figures,
// stars) driven by a handful of uniforms. Nothing here schedules frames; RealSky does.
import type { SkyCatalog } from "./catalog";
import type { Camera } from "./view";
import { LINE_FS, LINE_VS, SKY_FS, SKY_VS, STAR_FS, STAR_VS } from "./shaders";
import { mulMat, toColumnMajor, type Mat3 } from "./vec";

export interface SkyFrame {
  /** J2000 equatorial → horizon */
  rot: Mat3;
  cam: Camera;
  /** constellation drawn in amber, −1 for none */
  figure: number;
  /** figure line opacity and width (px) */
  lineAlpha: number;
  lineWidth: number;
  /** star size, px per unit */
  starScale: number;
}

export interface SkyRenderer {
  draw(f: SkyFrame): void;
  /** true between webglcontextlost and webglcontextrestored */
  readonly lost: boolean;
  dispose(): void;
}

/** Brightness of stars and figure lines below the horizon. */
const BELOW = 0.36;

type Prog = { p: WebGLProgram; u: (name: string) => WebGLUniformLocation | null; a: (name: string) => number };

function program(gl: WebGLRenderingContext, vs: string, fs: string): Prog {
  const shader = (type: number, src: string) => {
    const s = gl.createShader(type)!;
    gl.shaderSource(s, src);
    gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS) && !gl.isContextLost()) throw new Error(`sky shader: ${gl.getShaderInfoLog(s)}`);
    return s;
  };
  const p = gl.createProgram()!;
  gl.attachShader(p, shader(gl.VERTEX_SHADER, vs));
  gl.attachShader(p, shader(gl.FRAGMENT_SHADER, fs));
  gl.linkProgram(p);
  if (!gl.getProgramParameter(p, gl.LINK_STATUS) && !gl.isContextLost()) throw new Error(`sky program: ${gl.getProgramInfoLog(p)}`);
  const us = new Map<string, WebGLUniformLocation | null>();
  return { p, u: (n) => (us.has(n) ? us.get(n)! : (us.set(n, gl.getUniformLocation(p, n)), us.get(n)!)), a: (n) => gl.getAttribLocation(p, n) };
}

/** Interleaved per-star data: x, y, z, magnitude, B−V. */
function starData(cat: SkyCatalog): Float32Array<ArrayBuffer> {
  const n = cat.starMag.length, out = new Float32Array(n * 5);
  for (let i = 0; i < n; i++) {
    out.set(cat.starDirs.subarray(i * 3, i * 3 + 3), i * 5);
    out[i * 5 + 3] = cat.starMag[i]!;
    out[i * 5 + 4] = cat.starBv[i]!;
  }
  return out;
}

/** Four corners per segment (a, b, which end, side, constellation) and two triangles. */
function lineData(cat: SkyCatalog): { verts: Float32Array<ArrayBuffer>; index: Uint16Array<ArrayBuffer> } {
  const n = cat.segFigure.length, verts = new Float32Array(n * 4 * 9), index = new Uint16Array(n * 6);
  const dir = (v: number) => cat.lineDirs.subarray(v * 3, v * 3 + 3);
  for (let i = 0; i < n; i++) {
    const a = dir(cat.segments[i * 2]!), b = dir(cat.segments[i * 2 + 1]!), fig = cat.segFigure[i]!;
    // a+, a−, b−, b+ : around the quad
    ([[0, 1], [0, -1], [1, -1], [1, 1]] as const).forEach(([end, side], k) => {
      const o = (i * 4 + k) * 9;
      verts.set(a, o);
      verts.set(b, o + 3);
      verts[o + 6] = end;
      verts[o + 7] = side;
      verts[o + 8] = fig;
    });
    index.set([i * 4, i * 4 + 1, i * 4 + 2, i * 4, i * 4 + 2, i * 4 + 3], i * 6);
  }
  return { verts, index };
}

/** null when WebGL is unavailable or the shaders do not compile (the caller falls back). */
export function createSkyRenderer(canvas: HTMLCanvasElement, cat: SkyCatalog, onLost: (lost: boolean) => void): SkyRenderer | null {
  const gl = canvas.getContext("webgl", { alpha: false, antialias: false, depth: false, stencil: false, premultipliedAlpha: true, powerPreference: "low-power" });
  if (!gl) return null;
  const stars = starData(cat), lines = lineData(cat), m = new Float32Array(9);
  let sky: Prog, star: Prog, line: Prog;
  let triBuf: WebGLBuffer, starBuf: WebGLBuffer, lineBuf: WebGLBuffer, lineIdx: WebGLBuffer;
  let lost = false;

  const setup = () => {
    sky = program(gl, SKY_VS, SKY_FS);
    star = program(gl, STAR_VS, STAR_FS);
    line = program(gl, LINE_VS, LINE_FS);
    const buffer = (target: number, data: BufferSource) => {
      const b = gl.createBuffer()!;
      gl.bindBuffer(target, b);
      gl.bufferData(target, data, gl.STATIC_DRAW);
      return b;
    };
    triBuf = buffer(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 3, -1, -1, 3]));
    starBuf = buffer(gl.ARRAY_BUFFER, stars);
    lineBuf = buffer(gl.ARRAY_BUFFER, lines.verts);
    lineIdx = buffer(gl.ELEMENT_ARRAY_BUFFER, lines.index);
    gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA);
  };
  try {
    setup();
  } catch (err) {
    console.warn("[sky]", err);
    return null;
  }

  const onContextLost = (e: Event) => {
    e.preventDefault();
    lost = true;
    onLost(true);
  };
  const onContextRestored = () => {
    try {
      setup();
      lost = false;
      onLost(false);
    } catch (err) {
      console.warn("[sky]", err);
    }
  };
  canvas.addEventListener("webglcontextlost", onContextLost);
  canvas.addEventListener("webglcontextrestored", onContextRestored);

  const attrib = (pr: Prog, name: string, size: number, stride: number, offset: number) => {
    const loc = pr.a(name);
    gl.enableVertexAttribArray(loc);
    gl.vertexAttribPointer(loc, size, gl.FLOAT, false, stride, offset);
    return loc;
  };
  const common = (pr: Prog, f: SkyFrame) => {
    gl.uniformMatrix3fv(pr.u("u_m"), false, m);
    gl.uniform2f(pr.u("u_res"), f.cam.w, f.cam.h);
    gl.uniform2f(pr.u("u_c"), f.cam.cx, f.cam.cy);
    gl.uniform1f(pr.u("u_k"), f.cam.k);
  };

  return {
    get lost() {
      return lost;
    },
    draw(f) {
      if (lost) return;
      const { cam } = f;
      if (canvas.width !== cam.w || canvas.height !== cam.h) {
        canvas.width = cam.w;
        canvas.height = cam.h;
      }
      gl.viewport(0, 0, cam.w, cam.h);
      toColumnMajor(mulMat([...cam.r, ...cam.u, ...cam.f] as Mat3, f.rot), m);
      const hz = [f.rot[6], f.rot[7], f.rot[8]] as const;

      gl.disable(gl.BLEND);
      gl.useProgram(sky.p);
      gl.bindBuffer(gl.ARRAY_BUFFER, triBuf);
      const pos = attrib(sky, "a_pos", 2, 0, 0);
      gl.uniform2f(sky.u("u_res"), cam.w, cam.h);
      gl.uniform2f(sky.u("u_c"), cam.cx, cam.cy);
      gl.uniform1f(sky.u("u_k"), cam.k);
      gl.uniform3f(sky.u("u_cup"), cam.r[2], cam.u[2], cam.f[2]);
      gl.drawArrays(gl.TRIANGLES, 0, 3);
      gl.disableVertexAttribArray(pos);

      gl.enable(gl.BLEND);
      gl.useProgram(line.p);
      common(line, f);
      gl.uniform3f(line.u("u_hz"), ...hz);
      gl.uniform1f(line.u("u_width"), f.lineWidth);
      gl.uniform1f(line.u("u_fig"), f.figure);
      gl.uniform1f(line.u("u_below"), BELOW);
      const a = f.lineAlpha;
      gl.uniform4f(line.u("u_color"), (150 / 255) * a, (172 / 255) * a, a, a);
      gl.uniform4f(line.u("u_hiColor"), 0.99 * 0.9, 0.83 * 0.9, 0.3 * 0.9, 0.9);
      gl.bindBuffer(gl.ARRAY_BUFFER, lineBuf);
      gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, lineIdx);
      const l1 = attrib(line, "a_a", 3, 36, 0), l2 = attrib(line, "a_b", 3, 36, 12), l3 = attrib(line, "a_meta", 3, 36, 24);
      gl.drawElements(gl.TRIANGLES, lines.index.length, gl.UNSIGNED_SHORT, 0);
      [l1, l2, l3].forEach((l) => gl.disableVertexAttribArray(l));

      gl.useProgram(star.p);
      common(star, f);
      gl.uniform3f(star.u("u_hz"), ...hz);
      gl.uniform1f(star.u("u_scale"), f.starScale);
      gl.uniform1f(star.u("u_below"), BELOW);
      gl.bindBuffer(gl.ARRAY_BUFFER, starBuf);
      const s1 = attrib(star, "a_dir", 3, 20, 0), s2 = attrib(star, "a_magbv", 2, 20, 12);
      gl.drawArrays(gl.POINTS, 0, cat.starMag.length);
      [s1, s2].forEach((l) => gl.disableVertexAttribArray(l));
    },
    dispose() {
      canvas.removeEventListener("webglcontextlost", onContextLost);
      canvas.removeEventListener("webglcontextrestored", onContextRestored);
      if (!lost) {
        [triBuf, starBuf, lineBuf, lineIdx].forEach((b) => gl.deleteBuffer(b));
        [sky, star, line].forEach((pr) => gl.deleteProgram(pr.p));
      }
      gl.getExtension("WEBGL_lose_context")?.loseContext();
    },
  };
}
```

```ts
      const x = cx + tx * s, y = cy + ty * s, ang = Math.atan2(ty, tx), L = 9 * dpr;

      ctx.closePath();
      ctx.fill();
      ctx.textAlign = "center";
      label(ctx, "Qibla", x - Math.cos(ang) * 22 * dpr, y - Math.sin(ang) * 14 * dpr, "rgba(252,211,77,.95)");
      bodies.push({ kind: "qibla", x, y });
    }
  }
  return bodies;
}
```

```ts
import { ref, shallowRef } from "vue";
import type { SkyPick } from "@/utils/sky/describe";
import type { ViewState } from "@/utils/sky/view";
import { clamp, wrapPi } from "@/utils/sky/vec";

export type { SkyPick };
/** What the real sky behind the page can show: SkyBackground falls back on anything but "ready". */
export type SkyStatus = "loading" | "ready" | "no-webgl" | "catalog-failed" | "lost";

const isOpen = ref(false);
/** The home sky is drawn and can be explored (kept by RealSky). */
const available = ref(false);
/** Where explore looks; replaced on every change so watchers fire. */
const view = shallowRef<ViewState>({ alt: 0, az: 0, fovDeg: 100 });
/** Where the view centre sits: 0 = on the orbit's Moon, 1 = the screen centre. */
const centre = ref(0);
/** The time slider's instant; null = now. */
const time = shallowRef<Date | null>(null);
const selection = shallowRef<SkyPick | null>(null);
/** The constellation under the mouse (desktop hover), −1 for none. */
const hoverFigure = ref(-1);
/** Bumped by RealSky when the sky can no longer be drawn while exploring. */
const closeRequests = ref(0);
/** The home background's current view, kept by RealSky; explore starts from it. Plain, not reactive. */
const backdrop: ViewState = { alt: 0, az: 0, fovDeg: 100 };

let glideFrame = 0;
const ease = (t: number) => t * t * (3 - 2 * t);

/** Animates the view (and the centre) to a target; a new glide, a drag or a flick cancels it. */
function glide(to: Partial<ViewState> & { centre?: number }, ms = 500, done?: () => void) {
  cancelAnimationFrame(glideFrame);
  const from = view.value, c0 = centre.value, t0 = performance.now();
  const dAz = wrapPi((to.az ?? from.az) - from.az);
  const step = (now: number) => {
    const t = ease(clamp((now - t0) / ms, 0, 1));
    view.value = {
      alt: from.alt + ((to.alt ?? from.alt) - from.alt) * t,
      az: from.az + dAz * t,
      fovDeg: from.fovDeg + ((to.fovDeg ?? from.fovDeg) - from.fovDeg) * t,
    };
    centre.value = c0 + ((to.centre ?? c0) - c0) * t;
    if (t < 1) glideFrame = requestAnimationFrame(step);
    else {
      glideFrame = 0;
      done?.();
    }
  };
  glideFrame = requestAnimationFrame(step);
}

function stopGlide() {
  cancelAnimationFrame(glideFrame);
  glideFrame = 0;
}

/** Explore mode's state for this window: RealSky draws it, sky/Explore.vue drives it. */
export function useSkyExplore() {
  return { isOpen, available, view, centre, time, selection, hoverFigure, closeRequests, backdrop, glide, stopGlide };
}
```

```vue
// ── Explore input: drag (with a glide after a flick), pinch and wheel zoom, tap, hover ─────────
const pointers = new Map<number, { x: number; y: number }>();
let gesture: { x0: number; y0: number; moved: boolean; t: number; vx: number; vy: number } | null = null;
let pinch = 0;
let inertiaFrame = 0;

function spread() {
  const [a, b] = [...pointers.values()];
  return a && b ? Math.hypot(a.x - b.x, a.y - b.y) : 0;
}

function onDown(e: PointerEvent) {
  if (!props.interactive) return;
  textCanvas.value!.setPointerCapture(e.pointerId);
  pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
  explore.stopGlide();
  cancelAnimationFrame(inertiaFrame);
  if (pointers.size === 1) gesture = { x0: e.clientX, y0: e.clientY, moved: false, t: e.timeStamp, vx: 0, vy: 0 };
  else {
    pinch = spread();
    if (gesture) gesture.moved = true;
  }
}

function onMove(e: PointerEvent) {
  if (!props.interactive) return;
  const prev = pointers.get(e.pointerId);
  if (!prev) {
    if (e.pointerType === "mouse") hover(e);
    return;
  }
  pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
  if (pointers.size >= 2) {
    const d = spread(), [a, b] = [...pointers.values()];
    if (pinch > 0 && d > 0 && a && b) zoomBy(pinch / d, (a.x + b.x) / 2, (a.y + b.y) / 2);
    pinch = d;
    return;
  }
  if (!gesture) return;
  const dx = e.clientX - prev.x, dy = e.clientY - prev.y;
  if (!gesture.moved && Math.hypot(e.clientX - gesture.x0, e.clientY - gesture.y0) > 6) gesture.moved = true;
  const dt = Math.max(1, e.timeStamp - gesture.t);
  gesture.vx = 0.8 * (dx / dt) + 0.2 * gesture.vx;
  gesture.vy = 0.8 * (dy / dt) + 0.2 * gesture.vy;
  gesture.t = e.timeStamp;
  explore.view.value = pan(explore.view.value, dx, dy, cssWidth());
}

function onUp(e: PointerEvent) {
  if (!pointers.delete(e.pointerId)) return;
  if (pointers.size > 0) {
    pinch = spread();
    return;
  }
  const g = gesture;
  gesture = null;
  pinch = 0;
  if (!g || !props.interactive) return;
  if (!g.moved) tap(e);
  else if (e.timeStamp - g.t < 80) glideOn(g.vx, g.vy);
}

function onLeave(e: PointerEvent) {
  if (e.pointerType === "mouse" && !pointers.size) explore.hoverFigure.value = -1;
}

/** Keeps drifting after a flick, slowing to a stop in about 0.6 s. */
function glideOn(vx: number, vy: number) {
  if (Math.hypot(vx, vy) < 0.05) return;
  let last = performance.now();
  const step = (now: number) => {
    const dt = now - last, k = Math.exp(-dt / 200);
    last = now;
    vx *= k;
    vy *= k;
    explore.view.value = pan(explore.view.value, vx * dt, vy * dt, cssWidth());
    inertiaFrame = Math.hypot(vx, vy) > 0.02 ? requestAnimationFrame(step) : 0;
  };
  inertiaFrame = requestAnimationFrame(step);
}

function onWheel(e: WheelEvent) {
  if (props.interactive) zoomBy(Math.exp(e.deltaY * 0.0012), e.clientX, e.clientY);
}

/** Zoom around a point given in client coordinates. */
function zoomBy(factor: number, clientX: number, clientY: number) {
  if (!lastCam || !root.value) return;
  explore.stopGlide();
  const base = root.value.getBoundingClientRect(), dpr = dprNow();
  explore.view.value = zoomAt(explore.view.value, factor, (clientX - base.left) * dpr, (clientY - base.top) * dpr, lastCam.cx, lastCam.cy, lastCam.w, lastCam.h);
}

function hitAt(clientX: number, clientY: number): SkyPick | null {
  if (!lastCam || !lastSky || !cat || !root.value) return null;
  const base = root.value.getBoundingClientRect(), dpr = dprNow();
  if (starXY.length !== cat.starMag.length * 2) starXY = new Float32Array(cat.starMag.length * 2);
  if (vertXY.length !== (cat.lineDirs.length / 3) * 2) vertXY = new Float32Array((cat.lineDirs.length / 3) * 2);
  projectAll(lastCam, lastSky.rot, cat.starDirs, starXY);
  projectAll(lastCam, lastSky.rot, cat.lineDirs, vertXY);
  const hit = pick({
    x: (clientX - base.left) * dpr,
    y: (clientY - base.top) * dpr,
    bodies,
    starXY,
    starMag: cat.starMag,
    vertXY,
    segments: cat.segments,
    radius: 11 * dpr,
    segRadius: 7 * dpr,
  });
  if (!hit) return null;
  if (hit.kind === "star") return { kind: "star", index: hit.index };
  if (hit.kind === "segment") return { kind: "figure", figure: cat.segFigure[hit.index]! };
  const b = bodies[hit.index]!;
  return b.kind === "planet" ? { kind: "planet", key: b.key } : { kind: b.kind };
}

function tap(e: PointerEvent) {
  const p = hitAt(e.clientX, e.clientY);
  explore.selection.value = p;
  if (p?.kind === "qibla" && lastSky && cat) {
    const { alt, az } = altAzOf(directionOf(p, lastSky, cat, props.lat, props.lng));
    explore.glide({ alt, az });
  }
}

let hoverQueued = false;
function hover(e: PointerEvent) {
  if (hoverQueued) return;
  hoverQueued = true;
  const { clientX, clientY } = e;
  requestAnimationFrame(() => {
    hoverQueued = false;
    if (!cat || !props.interactive) return;
    const p = hitAt(clientX, clientY), fig = figureOf(p, cat);
    if (fig !== explore.hoverFigure.value) explore.hoverFigure.value = fig;
    if (textCanvas.value) textCanvas.value.style.cursor = p ? "pointer" : "";
  });
}
</script>
```

```vue
  // Esc clears the query first; with nothing typed it leaves the field (the next Esc closes explore).
  else if (e.key === "Escape") {
    if (query.value) query.value = "";
    else (e.target as HTMLElement).blur();
  } else return;
  e.preventDefault();
}
```

```vue
function open() {

  explore.view.value = { alt: b.alt, az: b.az, fovDeg: b.fovDeg };
  explore.centre.value = 0;
  time.value = null;
  selection.value = null;
  isOpen.value = true;

function close() {
  time.value = null;
  selection.value = null;
  explore.hoverFigure.value = -1;

  });

  if (window.history.state?.back) router.back();
  else void navigateTo({ query: { ...route.query, sky: undefined } }, { replace: true });

  if (!isOpen.value) return;
  explore.stopGlide();

  time.value = null;
  selection.value = null;

  const d = getNow();
  d.setHours(0, 0, 0, 0);

  const [h, m] = nowHHMM.value.split(":").map(Number);

const ARROWS: Record<string, [number, number]> = { ArrowLeft: [40, 0], ArrowRight: [-40, 0], ArrowUp: [0, 40], ArrowDown: [0, -40] };
useEventListener(window, "keydown", (e: KeyboardEvent) => {

  else if (step) {
    explore.stopGlide();
    explore.view.value = pan(v, step[0], step[1], window.innerWidth);
  } else if (e.key === "+" || e.key === "=") explore.view.value = { ...v, fovDeg: Math.max(FOV_MIN, v.fovDeg * 0.8) };
  else if (e.key === "-" || e.key === "_") explore.view.value = { ...v, fovDeg: Math.min(FOV_MAX, v.fovDeg * 1.25) };
  else return;
  e.preventDefault();
});
```
